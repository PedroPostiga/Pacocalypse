10// Finds the pixel position of the TILE_PLAYER_SPAWN tile in the map.
11// Sets (x, y) to the top-left corner of that tile.
12// Returns true on success, false if no spawn tile is found.
13static bool find_spawn(const map_t *map, int *out_x, int *out_y) {
14 for (int row = 0; row < MAP_ROWS; row++) {
15 for (int col = 0; col < MAP_COLS; col++) {
16 if (map->tiles[row][col].type == TILE_PLAYER_SPAWN) {
17 *out_x = MAP_OFFSET_X + col * TILE_SIZE;
18 *out_y = MAP_OFFSET_Y + row * TILE_SIZE;
19 return true;
20 }
21 }
22 }
23 return false;
24}

26// Returns the pixel delta (dx, dy) for one step in the given direction.
27static void direction_to_delta(direction_t dir, int *dx, int *dy) {
28 *dx = 0;
29 *dy = 0;
30 switch (dir) {
31 case DIR_UP: *dy = -PLAYER_SPEED; break;
32 case DIR_DOWN: *dy = PLAYER_SPEED; break;
33 case DIR_LEFT: *dx = -PLAYER_SPEED; break;
34 case DIR_RIGHT: *dx = PLAYER_SPEED; break;
35 default: break;
36 }

62// Returns true if the player at position (px, py) can move without entering a wall.
63// Checks all four corners of the player sprite (TILE_SIZE x TILE_SIZE bounding box).
64static bool can_move_to(const map_t *map, int px, int py) {
65 // Four corners of the bounding box (shrink by 1 to avoid edge collisions on border)
66 int corners[4][2] = {
67 { px + 1, py + 1 }, // top-left
68 { px + TILE_SIZE - 2, py + 1 }, // top-right
69 { px + 1, py + TILE_SIZE - 2 }, // bottom-left
70 { px + TILE_SIZE - 2, py + TILE_SIZE - 2 }, // bottom-right
71 };
72
73 for (int i = 0; i < 4; i++) {
74 int col, row;
75 map_tile_from_pixel(corners[i][0], corners[i][1], &row, &col);
76 if (!map_is_walkable(map, row, col))
77 return false;
78 }
79 return true;
80}

82static bool player_is_fully_in_single_tile(const player_t *player) {
83 int corners[4][2] = {
84 { player->x + 1, player->y + 1 },
85 { player->x + TILE_SIZE - 2, player->y + 1 },
86 { player->x + 1, player->y + TILE_SIZE - 2 },
87 { player->x + TILE_SIZE - 2, player->y + TILE_SIZE - 2 },
88 };
89
90 int first_col = (corners[0][0] - MAP_OFFSET_X) / TILE_SIZE;
91 int first_row = (corners[0][1] - MAP_OFFSET_Y) / TILE_SIZE;
92
93 for (int i = 1; i < 4; i++) {
94 int col = (corners[i][0] - MAP_OFFSET_X) / TILE_SIZE;
95 int row = (corners[i][1] - MAP_OFFSET_Y) / TILE_SIZE;
96
97 if (col != first_col || row != first_row)
98 return false;
99 }
100
101 return true;
102}

142void player_move(player_t *player, map_t *map) {
143 if (!player->alive) return;
144
145 int dx, dy;
146
147 // Try the buffered direction only at clean tile positions.
148 if (player->next_direction != DIR_NONE && player_is_fully_in_single_tile(player)) {
149 if (player_try_portal_wrap(player, player->next_direction)) {
150 player->direction = player->next_direction;
151 player->next_direction = DIR_NONE;
152 return;
153 }
154
155 direction_to_delta(player->next_direction, &dx, &dy);
156 int nx = player->x + dx;
157 int ny = player->y + dy;
158
159 if (can_move_to(map, nx, ny)) {
160 player->direction = player->next_direction;
161 player->next_direction = DIR_NONE;
162 player->x = nx;
163 player->y = ny;
164 return;
165 }
166 }
167
168 // Fall back to current direction
169 if (player->direction == DIR_NONE) return;
170
171 if (player_try_portal_wrap(player, player->direction))
172 return;
173
174 direction_to_delta(player->direction, &dx, &dy);
175 int nx = player->x + dx;
176 int ny = player->y + dy;
177
178 if (can_move_to(map, nx, ny)) {
179 player->x = nx;
180 player->y = ny;
181 }
