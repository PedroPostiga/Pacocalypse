40static void game_activate_power_up_at(game_state_t* state, int mouse_x, int mouse_y) {
41 if (!state || !state->player || !state->player->powered_up) return;
42
43 int radius_squared = POWER_UP_CLICK_RADIUS * POWER_UP_CLICK_RADIUS;
44 bool hit_ghost = false;
45
46 for (int i = 0; i < state->num_ghosts; i++) {
47 ghost_t *ghost = state->ghosts[i];
48 if (!ghost || ghost->state == GHOST_DEAD || ghost->state == GHOST_RESPAWNING) continue;
