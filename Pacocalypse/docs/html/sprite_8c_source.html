70animated_sprite_t *animated_sprite_create(xpm_map_t xpms[], uint8_t num_frames, uint8_t ticks_per_frame, uint16_t x, uint16_t y) {
71 animated_sprite_t *anim = malloc(sizeof(animated_sprite_t));
72 if (!anim) return NULL;
73
74 anim->frames = malloc(num_frames * sizeof(sprite_t*));
75 if (!anim->frames) {
76 free(anim);
77 return NULL;
78 }
79
80 for (int i = 0; i < num_frames; i++) {
81 anim->frames[i] = sprite_create(xpms[i]);
82 if (!anim->frames[i]) {
83 // Clean up previously created frames
84 for (int j = 0; j < i; j++) {
85 sprite_destroy(anim->frames[j]);
86 }
87 free(anim->frames);
88 free(anim);
89 return NULL;
90 }
91 }
92
93 anim->no_pixmaps = num_frames;
94 anim->current_pixmap = 0;
95 anim->ticks_per_frame = ticks_per_frame;
96 anim->tick_count = 0;
97 anim->x = x;
98 anim->y = y;
99
100 return anim;
101}
