117static void renderer_draw_ghost(ghost_t* const ghosts[GHOST_COUNT], const game_sprites_t *sprites) {
118 for (int i = 0; i < GHOST_COUNT; i++) {
119 ghost_t *ghost = ghosts[i];
120 if (!ghost) continue;
121
122 // Skip drawing if ghost is permanently dead (not respawning)
123 if (ghost->state == GHOST_DEAD) {
124 // Still need to draw eyes
125 }
126
127 animated_sprite_t *ghost_anim = renderer_select_ghost_sprite(ghost, sprites);
128 if (!ghost_anim) continue;
129
130 // For eyes sprites, set the frame based on direction
131 if (ghost->state == GHOST_DEAD || ghost->state == GHOST_RESPAWNING) {
132 int eyes_frame = renderer_get_eyes_frame(ghost->direction);
133 if (eyes_frame >= 0 && eyes_frame < (int)ghost_anim->no_pixmaps) {
134 ghost_anim->current_pixmap = eyes_frame;
135 }
136 }
137
138 sprite_t *frame = animated_sprite_get_current_frame(ghost_anim);
139 draw_sprite(frame, ghost->x, ghost->y);
140 }
141}

188void renderer_update_animations(const game_state_t *state, const view_resources_t *resources) {
189 if (!state || !resources || !resources->sprites) return;
190
191 game_sprites_t *sp = resources->sprites;
192
193 if (sp->player_anim_up) animated_sprite_update(sp->player_anim_up);
194 if (sp->player_anim_down) animated_sprite_update(sp->player_anim_down);
195 if (sp->player_anim_left) animated_sprite_update(sp->player_anim_left);
196 if (sp->player_anim_right) animated_sprite_update(sp->player_anim_right);
197
198 animated_sprite_t *ghost_sprites[GHOST_COUNT] = {
199 sp->ghost_red,
200 sp->ghost_pink,
201 sp->ghost_cyan,
202 sp->ghost_orange
203 };
204
205 for (int i = 0; i < state->num_ghosts; i++) {
206 ghost_t *ghost = state->ghosts[i];
207 if (!ghost) continue;
208
209 animated_sprite_t *anim = ghost_sprites[ghost->id];
210 if (anim) {
211 anim->current_pixmap = direction_to_ghost_frame(ghost->direction);
212 }
213 }
214
215 if (sp->frightened_ghost_anim) animated_sprite_update(sp->frightened_ghost_anim);
216 if (sp->ghost_eyes) animated_sprite_update(sp->ghost_eyes);
217}

219void renderer_draw_game(const game_state_t* state,
220 const view_resources_t *resources,
221 int mouse_x,

261int draw_sprite(const sprite_t* sprite, int x, int y) {
262 if (!sprite) return -1;
263
264 uint32_t *color = (uint32_t *) sprite->pixmap;
265 uint32_t transparent = xpm_transparency_color(XPM_8_8_8_8);
266
267 for (int j = 0; j < sprite->height; j++) {
268 for (int i = 0; i < sprite->width; i++) {
269 uint32_t pixel = color[j * sprite->width + i];
270 if (pixel != transparent) {
271 vg_draw_pixel(x + i, y + j, pixel);
272 }
273 }
274 }
275 return 0;
276}
